rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }

    function usuarioPath(uid) {
      return /databases/$(database)/documents/usuarios/$(uid);
    }

    function convitePath(email) {
      return /databases/$(database)/documents/convites_acesso/$(email);
    }

    function temPerfil() {
      return logado() && exists(usuarioPath(request.auth.uid));
    }

    function meuUsuario() {
      return get(usuarioPath(request.auth.uid));
    }

    function souAdmin() {
      return temPerfil()
        && meuUsuario().data.tipo == "admin"
        && meuUsuario().data.ativo == true;
    }

    function souCliente() {
      return temPerfil()
        && meuUsuario().data.tipo == "cliente"
        && meuUsuario().data.ativo == true;
    }

    function minhaEmpresa() {
      return meuUsuario().data.empresaId;
    }

    function podeAtivarPerfilPorConvite(userId) {
      return logado()
        && request.auth.uid == userId
        && request.resource.data.email == request.auth.token.email
        && exists(convitePath(request.resource.data.email))
        && get(convitePath(request.resource.data.email)).data.ativo == true
        && get(convitePath(request.resource.data.email)).data.email == request.auth.token.email
        && get(convitePath(request.resource.data.email)).data.tipo == request.resource.data.tipo
        && get(convitePath(request.resource.data.email)).data.empresaId == request.resource.data.empresaId
        && request.resource.data.ativo == true;
    }

    function camposClienteRecurso() {
      return [
        "clientePdfUrl",
        "clientePdfNome",
        "clientePdfStoragePath",
        "clienteWordUrl",
        "clienteWordNome",
        "clienteWordStoragePath",
        "clienteLinkUrl",
        "clienteStatus",
        "clienteAtualizadoEm",
        "clienteAtualizadoPor"
      ];
    }

    function clienteAtualizaRecursoLiberado() {
      return souCliente()
        && resource.data.empresaId == minhaEmpresa()
        && resource.data.clientePodeEditar == true
        && request.resource.data.empresaId == resource.data.empresaId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(camposClienteRecurso())
        && request.resource.data.clienteAtualizadoPor == request.auth.uid
        && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(["clientePdfUrl", "clientePdfNome", "clientePdfStoragePath"]) || resource.data.clientePermitePdf == true)
        && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(["clienteWordUrl", "clienteWordNome", "clienteWordStoragePath"]) || resource.data.clientePermiteWord == true)
        && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(["clienteLinkUrl"]) || resource.data.clientePermiteLink == true);
    }

    match /usuarios/{userId} {
      allow get: if souAdmin() || (logado() && request.auth.uid == userId);
      allow list: if souAdmin();
      allow create: if souAdmin() || podeAtivarPerfilPorConvite(userId);
      allow update, delete: if souAdmin();
    }

    match /convites_acesso/{emailId} {
      allow get: if souAdmin() || logado();
      allow list: if souAdmin();
      allow create, update, delete: if souAdmin();
    }

    match /empresas/{empresaId} {
      allow get, list: if souAdmin() || (souCliente() && minhaEmpresa() == empresaId);
      allow create, update, delete: if souAdmin();
    }

    match /respostas_iso/{respostaId} {
      allow get: if souAdmin() || (
        souCliente()
        && respostaId.matches('^' + minhaEmpresa() + '_.*$')
      );

      allow list: if souAdmin() || (
        souCliente()
        && resource.data.empresaId == minhaEmpresa()
      );

      allow create: if souAdmin() || (
        souCliente()
        && request.resource.data.empresaId == minhaEmpresa()
        && request.resource.data.requisitoId in ["4_1", "4_2"]
      );

      allow update: if souAdmin() || (
        souCliente()
        && resource.data.empresaId == minhaEmpresa()
        && request.resource.data.empresaId == minhaEmpresa()
        && request.resource.data.requisitoId in ["4_1", "4_2"]
        && !(resource.data.status in ["concluido", "aprovado"])
      );

      allow delete: if souAdmin();
    }

    match /arquivos/{arquivoId} {
      allow read: if souAdmin() || (
        souCliente() && (
          (resource.data.categoria == "material_apoio" && resource.data.publico == true) ||
          (resource.data.empresaId == minhaEmpresa())
        )
      );

      allow create, update, delete: if souAdmin();
    }

    match /empresa_pastas/{pastaId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, update, delete: if souAdmin();
    }

    match /empresa_recursos/{recursoId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, delete: if souAdmin();
      allow update: if souAdmin() || clienteAtualizaRecursoLiberado();
    }

    // Processos/produtos cadastrados pelo administrador para apontamento de produção.
    match /empresa_produtos/{produtoId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, update, delete: if souAdmin();
    }

    // Funcionários cadastrados pelo administrador para cálculo de minutos e horas no apontamento.
    match /empresa_funcionarios/{funcionarioId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, update, delete: if souAdmin();
    }

    // Equipes/células do apontamento. Admin monta a equipe; cliente apenas usa no lançamento da própria empresa.
    match /empresa_equipes/{equipeId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, update, delete: if souAdmin();
    }

    // Apontamento de produção por empresa: o usuário escolhe processo/produto, quantidade, equipe e minutos reais.
    match /empresa_apontamentos/{apontamentoId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create: if souAdmin() || (
        souCliente()
        && request.resource.data.empresaId == minhaEmpresa()
        && request.resource.data.criadoPor == request.auth.uid
      );
      allow update: if souAdmin() || (
        souCliente()
        && resource.data.empresaId == minhaEmpresa()
        && request.resource.data.empresaId == minhaEmpresa()
        && resource.data.criadoPor == request.auth.uid
        && request.resource.data.atualizadoPor == request.auth.uid
      );
      allow delete: if souAdmin() || (
        souCliente()
        && resource.data.empresaId == minhaEmpresa()
        && resource.data.criadoPor == request.auth.uid
      );
    }

    // Diário de bordo por empresa: horas contratadas, atividades e saldo.
    match /diario_bordo_config/{empresaId} {
      allow read: if souAdmin() || (souCliente() && minhaEmpresa() == empresaId);
      allow create, update, delete: if souAdmin();
    }

    match /diario_bordo_atividades/{atividadeId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, update, delete: if souAdmin();
    }

    match /materiais_apoio/{materialId} {
      allow read: if souAdmin() || (
        souCliente() && (
          resource.data.publico == true ||
          resource.data.empresaId == minhaEmpresa()
        )
      );
      allow create, update, delete: if souAdmin();
    }

    match /anexos_empresa/{anexoId} {
      allow read: if souAdmin() || (souCliente() && resource.data.empresaId == minhaEmpresa());
      allow create, update, delete: if souAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
